cmake_minimum_required(VERSION 3.10)
project(MorseAnalyser VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_executable(morse-analyser morse-analyser.cpp)
add_executable(morse-encoder morse-encoder.cpp)
add_executable(gen_test_wav tests/gen_test_wav.cpp)
add_executable(test_roundtrip tests/test_roundtrip.cpp)
foreach(t morse-analyser morse-encoder gen_test_wav test_roundtrip)
  if(NOT MSVC)
    target_compile_options(${t} PRIVATE -Wall -Wextra)
  endif()
endforeach()

# Windows GUI (native Win32; not built elsewhere)
if(WIN32)
  enable_language(RC)
  add_executable(morse-gui WIN32 gui/morse-gui.cpp gui/morse-gui.rc)
  target_link_libraries(morse-gui PRIVATE comctl32 comdlg32 winmm shell32 gdi32 user32)
  if(MSVC)
    target_compile_options(morse-gui PRIVATE /utf-8)
    target_link_options(morse-gui PRIVATE /MANIFEST:NO)  # the manifest is embedded by the .rc
  else()
    target_compile_options(morse-gui PRIVATE -Wall -Wextra)
    target_link_options(morse-gui PRIVATE -static)
  endif()
  install(TARGETS morse-gui RUNTIME DESTINATION bin)
endif()

enable_testing()
add_test(NAME roundtrip COMMAND test_roundtrip WORKING_DIRECTORY ${CMAKE_BINARY_DIR})
add_test(NAME gen COMMAND gen_test_wav ${CMAKE_BINARY_DIR}/test.wav "SOS HI")
add_test(NAME analyse COMMAND morse-analyser ${CMAKE_BINARY_DIR}/test.wav --outdir ${CMAKE_BINARY_DIR})
set_tests_properties(analyse PROPERTIES DEPENDS gen PASS_REGULAR_EXPRESSION "=> SOS HI")
add_test(NAME encode COMMAND morse-encoder "THE QUICK BROWN FOX" -o ${CMAKE_BINARY_DIR}/stego.wav --hidden "Hi!!")
add_test(NAME analyse_hidden COMMAND morse-analyser ${CMAKE_BINARY_DIR}/stego.wav --outdir ${CMAKE_BINARY_DIR})
set_tests_properties(analyse_hidden PROPERTIES DEPENDS encode
  PASS_REGULAR_EXPRESSION "=> THE QUICK BROWN FOX.*Hidden carrier text: \"Hi!!\"")

install(TARGETS morse-analyser morse-encoder RUNTIME DESTINATION bin)
